module mux4x1_GL(
  input wire a,
  input wire b,
  input wire c,
  input wire d,
  input wire [1:0] s,
  output wire y
);
  wire or0;
  wire or1;
  wire or2;
  wire or3;
  and a0 (or0, a, !s[0], !s[1]);
  and a1 (or1, b, s[0],  !s[1]);
  and a2 (or2, c, !s[0], s[1]);
  and a3 (or3, d, s[0],  s[1]);
  or or0 (y, or0, or1, or2, or3);
endmodule

module Binary_Gray_DF(
  input wire [3:0] a,
  output wire [3:0] b,
);
  assign b[3] = a[3];
  assign b[2] = a[3] ^ a[2];
  assign b[1] = a[2] ^ a[1];
  assign b[0] = a[1] ^ a[0];
endmodule
===================================
module Binary_Gray_BH(
  input wire [3:0] a,
  output wire [3:0] b,
);
  always@(*)
    begin
      b[3] = a[3];
      b[2] = a[3] ^ a[2];
      b[1] = a[2] ^ a[1];
      b[0] = a[1] ^ a[0];
    end
endmodule

module Decoder2X4_BH(
  input wire [1:0] a,
  input wire en,
  output wire [3:0] b
);
  always@(*)
    begin
      if(en)
        begin
          case(a)
            2'b00: b[0] = 1'b1;
            2'b01: b[1] = 1'b1;
            2'b10: b[2] = 1'b1;
            2'b11: b[3] = 1'b1;
          endcase
        end
      end
    end
endmodule
